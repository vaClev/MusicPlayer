@startuml MusicDatabaseSchema

' Настройки отображения
hide circle
skinparam linetype ortho
skinparam packageStyle rectangle
left to right direction

' Основная сущность: музыкальный файл
entity MusicFile {
  * id : INT <<PK>>
  --
  * filename : TEXT
  * filepath : TEXT
  * title : TEXT
  * artistId : INT <<FK>>
  album : TEXT
  * genreId : INT <<FK>>
  year : INT
  * filesize : BIGINT
  * duration : INTERVAL
  * uploadDate : TIMESTAMP WITH TIME ZONE
  * FileExtensionId : INT <<FK>>
}

' Сущность исполнителя
entity Artist {
  * Id : INT <<PK>>
  --
  * Name : VARCHAR(200)
  Bio : TEXT
  Country : VARCHAR(100)
  FormedYear : INT
}

' Сущность жанра
entity Genre {
  * Id : INT <<PK>>
  --
  * Name : VARCHAR(100)
  Description : TEXT
}

' Сущность дополнительных файлов
entity ExtraFile {
  * Id : INT <<PK>>
  --
  * OriginalFileName : TEXT
  * StoredFileName : TEXT
  * FilePath : TEXT
  * Description : TEXT
  * FileType : INT
  * FileSize : BIGINT
  * UploadDate : TIMESTAMP WITH TIME ZONE
  * MusicFileId : INT <<FK>>
  * FileExtensionId : INT <<FK>>
}

' Сущность: расширения файлов
entity FileExtension {
  * Id : INT <<PK>>
  --
  * Extension : VARCHAR(10)
  * MimeType : VARCHAR(100)
  * Category : VARCHAR(20)
  Description : TEXT
}

' Определение связей
Artist ||--o{ MusicFile : "исполняет"
Genre ||--o{ MusicFile : "относится к"
MusicFile ||--o{ ExtraFile : "имеет"
FileExtension ||--o{ MusicFile : "используется в"
FileExtension ||--o{ ExtraFile : "используется в"

' Примечания и пояснения
note top of MusicFile
  Основная сущность - аудиофайл
  * - обязательные поля
  PK - первичный ключ
  FK - внешний ключ
  artistId → Artist.Id
  genreId → Genre.Id
  FileExtensionId → FileExtension.Id
end note

note top of Artist
  Исполнитель музыкального произведения
  Связь: один Artist → много MusicFile
  Name - уникальное имя исполнителя
end note

note top of Genre
  Музыкальный жанр
  Связь: один Genre → много MusicFile
  Name - уникальное название жанра
end note

note top of ExtraFile
  Дополнительные файлы (ноты, тексты)
  FK - внешний ключ к MusicFile.id
  FK2 - внешний ключ к FileExtension.id
  Связь: один MusicFile → много ExtraFile
end note

note top of FileExtension
  Справочник расширений файлов
  Extension: ".mp3", ".jpg", ".pdf"
  MimeType: "audio/mpeg", "image/jpeg"
  Один ко многим с MusicFile и ExtraFile
end note
