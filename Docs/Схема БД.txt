@startuml MusicDatabaseSchema

' Настройки отображения
hide circle
skinparam linetype ortho
skinparam packageStyle rectangle
left to right direction

' Основная сущность: музыкальный файл
entity MusicFile {
  * id : INT <<PK>>
  --
  * filename : TEXT
  * filepath : TEXT
  * title : TEXT
  * artist : TEXT
  album : TEXT
  year : INT
  genre : TEXT
  * filesize : BIGINT
  * duration : INTERVAL
  * uploadDate : TIMESTAMP WITH TIME ZONE
  * FileExtensionId : INT <<FK>>
}

' Сущность дополнительных файлов
entity ExtraFile {
  * Id : INT <<PK>>
  --
  * OriginalFileName : TEXT
  * StoredFileName : TEXT
  * FilePath : TEXT
  * Description : TEXT
  * FileType : INT
  * FileSize : BIGINT
  * UploadDate : TIMESTAMP WITH TIME ZONE
  * MusicFileId : INT <<FK>>
  * FileExtensionId : INT <<FK>>
}

' Cущность: расширения файлов
entity FileExtension {
  * Id : INT <<PK>>
  --
  * Extension : VARCHAR(10)
  * MimeType : VARCHAR(100)
  * Category : VARCHAR(20)
  Description : TEXT
}

' Определение связи один-ко-многим
MusicFile ||--o{ ExtraFile : "имеет"
FileExtension ||--o{ MusicFile 
FileExtension ||--o{ ExtraFile : "используется в"

' Примечания и пояснения
note top of MusicFile
  Основная сущность - аудиофайл
  * - обязательные поля
  PK - первичный ключ
  FK - внешний ключ к FileExtension
end note

note top of ExtraFile
  Дополнительные файлы (ноты, тексты)
  FK - внешний ключ к MusicFile.id
  FK2 - внешний ключ к FileExtension.id
  Связь: один MusicFile → много ExtraFile
end note

note top of FileExtension
  Справочник расширений файлов
  Extension: ".mp3", ".jpg", ".pdf"
  MimeType: "audio/mpeg", "image/jpeg"
  Один ко многим с MusicFile и ExtraFile
end note

' Блок с примерами данных
package "Пример данных" {
