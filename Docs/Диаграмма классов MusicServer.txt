' === 1. МОДЕЛИ ДАННЫХ (Entities) ===
package Models #ccccff {
    class MusicFile {
        +string Title
        +int? ArtistId {fk}
        +int? GenreId {fk}
        +ICollection<ExtraFile> ExtraFiles
    }

    class ExtraFile {
        +ExtraFileType FileType {enum}
        +MusicFile MusicFile {nav}
    }

    enum ExtraFileType {
        SheetMusic,     // Ноты
        Tabs,           // Табулатуры
        Lyrics,         // Текст песни
        Chords,         // Аккорды
        Image,          // Обложка/фото
        Other           // Прочее
    }

    class Artist {
        +string Name
        +string Bio
        +ICollection<MusicFile> MusicFiles
    }

    class Genre {
        +string Name
        +string Description
        +ICollection<MusicFile> MusicFiles
    }

    ExtraFile *-- ExtraFileType
    MusicFile "1" --o{ "0..*" ExtraFile : содержит >
    Artist "1" --o{ "0..*" MusicFile : has >
    Genre "1" --o{ "0..*" MusicFile : has >

        -IConfiguration _configuration

        -IUploadServiceFactory _uploadServiceFactory
        -IUploadService? _uploadService
    }

    interface IUploadServiceFactory {
        +Create(string, string[]) : IUploadService
    }

    class UploadServiceFactory {
    }

        +CreateFilePath(string) : string
