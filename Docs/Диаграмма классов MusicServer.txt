' === 1. МОДЕЛИ ДАННЫХ (Entities) ===
package "Models" #ccccff{
    class MusicFile {
        +int Id {pk}
        +string Filename
        +string Filepath
        +string Title
        +string Artist
        +string Album
        +int? Year
        +string Genre
        +long Filesize
        +TimeSpan Duration
        +DateTime UploadDate
        +ICollection<ExtraFile> ExtraFiles
    }

    class ExtraFile {
        +int Id {pk}
        +string OriginalFileName
        +string StoredFileName
        +string FilePath
        +string Description
        +ExtraFileType FileType {enum}
        +long FileSize
        +DateTime UploadDate
        +int MusicFileId {fk}
        +MusicFile MusicFile {nav}
    }

    enum ExtraFileType {
        SheetMusic,     // Ноты
        Tabs,           // Табулатуры
        Lyrics,         // Текст песни
        Chords,         // Аккорды
        Image,          // Обложка/фото
        Other           // Прочее
    }
    ExtraFile *--ExtraFileType
    ' Связь между моделями
    MusicFile "1" --o{ "0..*" ExtraFile : содержит >
}

' === 3. СЛОЙ ДАННЫХ (Data Layer) ===
package "Database" {
    class AppDbContext {
        +DbSet<MusicFile> MusicFiles
        +DbSet<ExtraFile> ExtraFiles
        +OnConfiguring(DbContextOptionsBuilder)
        +OnModelCreating(ModelBuilder)
    }

    class DbContext {
        #SaveChanges()
        #DbSet<T> Set<T>()
    }

    AppDbContext --|> DbContext : наследует
    AppDbContext --> MusicFile : возвращает
    AppDbContext --> ExtraFile : возвращает
}

' === 4. СЛОЙ СЕРВИСОВ (Business Logic) ===
package "Services" #ffdddd{
    interface IMusicService {
        +UploadAsync(CreateMusicFileDTO)
        +GetAllAsync()
        +GetMusicFileAsync(int)
        +DeleteMusicFileAsync(int)
    }

    class MusicService {
        -AppDbContext _context
        -IUploadServiceFactory _uploadFactory
        +MusicService(AppDbContext, IUploadServiceFactory)
        +UploadAsync(CreateMusicFileDTO) : Task<MusicFileDTO>
        +GetAllAsync() : Task<List<MusicFileDTO>>
        +GetMusicFileAsync(int) : Task<MusicFileDTO?>
        +DeleteMusicFileAsync(int) : Task<bool>
        -ConvertToDTOs
    }

    interface IExtraFileService {
        +UploadExtraFileAsync(int, UploadExtraFileDTO)
        +GetByMusicFileIdAsync(int)
        +DeleteExtraFileAsync(int)
    }

    class ExtraFileService {
        -AppDbContext _context
        -IUploadServiceFactory _uploadFactory
        +ExtraFileService(AppDbContext, IUploadServiceFactory)
        +UploadExtraFileAsync(int, UploadExtraFileDTO) : Task<ExtraFileDTO>
        +GetByMusicFileIdAsync(int) : Task<List<ExtraFileDTO>>
        +DeleteExtraFileAsync(int) : Task<bool>
        -ConvertToDTOs
    }
    DTOs -- ExtraFileService : возвращает
    DTOs -- MusicService  : возвращает
    interface IUploadServiceFactory {
        +CreateUploadService(string) : IUploadService
    }

    class UploadServiceFactory {
        +CreateUploadService(string) : IUploadService
    }

    IMusicService <|.. MusicService : реализует
    IExtraFileService <|.. ExtraFileService : реализует
    IUploadServiceFactory <|.. UploadServiceFactory : реализует

    MusicService o--> AppDbContext
    ExtraFileService o--> AppDbContext
    MusicService --> IUploadServiceFactory : использует
    ExtraFileService --> IUploadServiceFactory : использует

}
